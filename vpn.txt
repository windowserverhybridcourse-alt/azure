ENABLE THE SECURITY FEATURES
===================================

enable
sh version

license boot module c1900 technology-package securityk9
yes


exit
wr
reload

===========================================
LEFT SIDE ROUTER (HQ ROUTER)
==========================================
en

hostname HQ
no ip domain-lookup

=============================================================
LEFT SIDE ROUTER VPN CONFIG (HQ)
=============================================================
en
conf t

ip access-list extended mynetwork
permit ip 192.168.0.0 0.0.0.255 192.168.1.0 0.0.0.255
exit

crypto isakmp policy 10
encryption aes 256
authentication pre-share
group 5
exit

crypto isakmp key secret@123 address 2.2.2.1

crypto ipsec transform-set HQ-BR esp-aes 256 esp-sha-hmac

crypto map IPSEC-HQ 10 ipsec-isakmp
set peer 2.2.2.1
set pfs group5
set security-association lifetime seconds 86400
set transform-set HQ-BR
match address mynetwork
exit

interface GigabitEthernet0/1
crypto map IPSEC-HQ
end
wr
